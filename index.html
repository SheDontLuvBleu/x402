async function paidFetch(url, options = {}) {
  let response = await fetch(url, options);

  // The server tells us exactly how to pay.
  if (response.status === 402) {
    const payment = response.headers.get('PAYMENT-REQUIRED');
    const requirements = JSON.parse(atob(payment));

    // Sign with the agent wallet, never expose the key.
    const signature = await agentWallet.signPayment(requirements);
    response = await fetch(url, {
      ...options,
      headers: { ...options.headers,
        'PAYMENT-SIGNATURE': btoa(JSON.stringify(signature)) }
    });
  }
  return response.json();
}
